-- ============================================
-- AIRLINE DATABASE MANAGEMENT SYSTEM (DBMS)
-- Complete SQL Schema with Sample Data
-- ============================================

-- Drop existing database if exists and create new one
DROP DATABASE IF EXISTS airline_dbms;
CREATE DATABASE airline_dbms;
USE airline_dbms;

-- ============================================
-- TABLE: users
-- Stores login credentials and user roles
-- ============================================
CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    payment_pin VARCHAR(4) DEFAULT '0000',
    role ENUM('passenger', 'admin', 'crew') DEFAULT 'passenger',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP NULL,
    is_active BOOLEAN DEFAULT TRUE,
    INDEX idx_username (username),
    INDEX idx_email (email)
);

-- ============================================
-- TABLE: passengers
-- Detailed passenger information
-- ============================================
CREATE TABLE passengers (
    passenger_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    date_of_birth DATE NOT NULL,
    gender ENUM('Male', 'Female', 'Other') NOT NULL,
    nationality VARCHAR(50) NOT NULL,
    passport_number VARCHAR(20) UNIQUE NOT NULL,
    passport_expiry DATE NOT NULL,
    phone VARCHAR(20),
    address TEXT,
    frequent_flyer_number VARCHAR(20) UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_passport (passport_number)
);

-- ============================================
-- TABLE: airports
-- Airport information
-- ============================================
CREATE TABLE airports (
    airport_id INT PRIMARY KEY AUTO_INCREMENT,
    airport_code VARCHAR(3) UNIQUE NOT NULL,
    airport_name VARCHAR(100) NOT NULL,
    city VARCHAR(50) NOT NULL,
    country VARCHAR(50) NOT NULL,
    timezone VARCHAR(50),
    latitude DECIMAL(10, 6),
    longitude DECIMAL(10, 6),
    INDEX idx_airport_code (airport_code),
    INDEX idx_city (city)
);

-- ============================================
-- TABLE: aircraft
-- Aircraft fleet information
-- ============================================
CREATE TABLE aircraft (
    aircraft_id INT PRIMARY KEY AUTO_INCREMENT,
    aircraft_model VARCHAR(50) NOT NULL,
    registration_number VARCHAR(10) UNIQUE NOT NULL,
    manufacturer VARCHAR(50) NOT NULL,
    total_seats INT NOT NULL,
    economy_seats INT NOT NULL,
    business_seats INT NOT NULL,
    first_class_seats INT NOT NULL,
    year_manufactured YEAR,
    status ENUM('active', 'maintenance', 'retired') DEFAULT 'active',
    INDEX idx_registration (registration_number)
);

-- ============================================
-- TABLE: flights
-- Flight schedule information
-- ============================================
CREATE TABLE flights (
    flight_id INT PRIMARY KEY AUTO_INCREMENT,
    flight_number VARCHAR(10) UNIQUE NOT NULL,
    aircraft_id INT NOT NULL,
    departure_airport_id INT NOT NULL,
    arrival_airport_id INT NOT NULL,
    departure_time DATETIME NOT NULL,
    arrival_time DATETIME NOT NULL,
    duration_minutes INT NOT NULL,
    economy_price DECIMAL(10, 2) NOT NULL,
    business_price DECIMAL(10, 2) NOT NULL,
    first_class_price DECIMAL(10, 2) NOT NULL,
    available_economy_seats INT NOT NULL,
    available_business_seats INT NOT NULL,
    available_first_class_seats INT NOT NULL,
    status ENUM('scheduled', 'boarding', 'departed', 'arrived', 'delayed', 'cancelled') DEFAULT 'scheduled',
    gate_number VARCHAR(10),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (aircraft_id) REFERENCES aircraft(aircraft_id),
    FOREIGN KEY (departure_airport_id) REFERENCES airports(airport_id),
    FOREIGN KEY (arrival_airport_id) REFERENCES airports(airport_id),
    INDEX idx_flight_number (flight_number),
    INDEX idx_departure_time (departure_time),
    INDEX idx_status (status)
);

-- ============================================
-- TABLE: bookings
-- Flight booking records
-- ============================================
CREATE TABLE bookings (
    booking_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_reference VARCHAR(10) UNIQUE NOT NULL,
    passenger_id INT NOT NULL,
    flight_id INT NOT NULL,
    seat_number VARCHAR(5),
    class ENUM('economy', 'business', 'first_class') NOT NULL,
    booking_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total_amount DECIMAL(10, 2) NOT NULL,
    payment_status ENUM('pending', 'completed', 'failed', 'refunded') DEFAULT 'pending',
    booking_status ENUM('confirmed', 'cancelled', 'pending') DEFAULT 'pending',
    special_requests TEXT,
    baggage_allowance INT DEFAULT 20,
    meal_preference VARCHAR(50),
    FOREIGN KEY (passenger_id) REFERENCES passengers(passenger_id) ON DELETE CASCADE,
    FOREIGN KEY (flight_id) REFERENCES flights(flight_id) ON DELETE CASCADE,
    INDEX idx_booking_ref (booking_reference),
    INDEX idx_passenger (passenger_id),
    INDEX idx_flight (flight_id),
    INDEX idx_booking_date (booking_date)
);

-- ============================================
-- TABLE: payments
-- Payment transaction records
-- ============================================
CREATE TABLE payments (
    payment_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    payment_method ENUM('credit_card', 'debit_card', 'paypal', 'bank_transfer', 'cash') NOT NULL,
    amount DECIMAL(10, 2) NOT NULL,
    currency VARCHAR(3) DEFAULT 'USD',
    transaction_id VARCHAR(100) UNIQUE,
    payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    payment_status ENUM('success', 'failed', 'pending', 'refunded') DEFAULT 'pending',
    card_last_four VARCHAR(4),
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE CASCADE,
    INDEX idx_booking (booking_id),
    INDEX idx_transaction (transaction_id)
);

-- ============================================
-- TABLE: frequent_flyer_miles
-- Loyalty program miles tracking
-- ============================================
CREATE TABLE frequent_flyer_miles (
    miles_id INT PRIMARY KEY AUTO_INCREMENT,
    passenger_id INT NOT NULL,
    total_miles INT DEFAULT 0,
    tier ENUM('Silver', 'Gold', 'Platinum', 'Diamond') DEFAULT 'Silver',
    miles_earned INT DEFAULT 0,
    miles_redeemed INT DEFAULT 0,
    lifetime_miles INT DEFAULT 0,
    tier_valid_until DATE,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (passenger_id) REFERENCES passengers(passenger_id) ON DELETE CASCADE,
    INDEX idx_passenger (passenger_id),
    INDEX idx_tier (tier)
);

-- ============================================
-- TABLE: miles_transactions
-- Miles earned/redeemed history
-- ============================================
CREATE TABLE miles_transactions (
    transaction_id INT PRIMARY KEY AUTO_INCREMENT,
    passenger_id INT NOT NULL,
    booking_id INT,
    transaction_type ENUM('earned', 'redeemed', 'expired', 'bonus') NOT NULL,
    miles_amount INT NOT NULL,
    description VARCHAR(255),
    transaction_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (passenger_id) REFERENCES passengers(passenger_id) ON DELETE CASCADE,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE SET NULL,
    INDEX idx_passenger (passenger_id),
    INDEX idx_date (transaction_date)
);

-- ============================================
-- TABLE: offers
-- Special offers and promotions
-- ============================================
CREATE TABLE offers (
    offer_id INT PRIMARY KEY AUTO_INCREMENT,
    offer_code VARCHAR(20) UNIQUE NOT NULL,
    offer_name VARCHAR(100) NOT NULL,
    description TEXT,
    discount_type ENUM('percentage', 'fixed_amount') NOT NULL,
    discount_value DECIMAL(10, 2) NOT NULL,
    valid_from DATE NOT NULL,
    valid_until DATE NOT NULL,
    max_usage INT,
    current_usage INT DEFAULT 0,
    min_purchase_amount DECIMAL(10, 2),
    applicable_class ENUM('all', 'economy', 'business', 'first_class') DEFAULT 'all',
    is_active BOOLEAN DEFAULT TRUE,
    category VARCHAR(50),
    INDEX idx_offer_code (offer_code),
    INDEX idx_valid_dates (valid_from, valid_until)
);

-- ============================================
-- TABLE: offer_usage
-- Track who used which offers
-- ============================================
CREATE TABLE offer_usage (
    usage_id INT PRIMARY KEY AUTO_INCREMENT,
    offer_id INT NOT NULL,
    booking_id INT NOT NULL,
    passenger_id INT NOT NULL,
    discount_applied DECIMAL(10, 2) NOT NULL,
    used_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (offer_id) REFERENCES offers(offer_id) ON DELETE CASCADE,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE CASCADE,
    FOREIGN KEY (passenger_id) REFERENCES passengers(passenger_id) ON DELETE CASCADE,
    INDEX idx_offer (offer_id),
    INDEX idx_passenger (passenger_id)
);

-- ============================================
-- TABLE: crew
-- Flight crew information
-- ============================================
CREATE TABLE crew (
    crew_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    role ENUM('pilot', 'co-pilot', 'flight_attendant', 'purser') NOT NULL,
    license_number VARCHAR(20) UNIQUE,
    hire_date DATE NOT NULL,
    date_of_birth DATE NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100),
    status ENUM('active', 'on_leave', 'retired') DEFAULT 'active',
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE SET NULL,
    INDEX idx_role (role),
    INDEX idx_license (license_number)
);

-- ============================================
-- TABLE: flight_crew_assignment
-- Assign crew members to flights
-- ============================================
CREATE TABLE flight_crew_assignment (
    assignment_id INT PRIMARY KEY AUTO_INCREMENT,
    flight_id INT NOT NULL,
    crew_id INT NOT NULL,
    assigned_role VARCHAR(50),
    FOREIGN KEY (flight_id) REFERENCES flights(flight_id) ON DELETE CASCADE,
    FOREIGN KEY (crew_id) REFERENCES crew(crew_id) ON DELETE CASCADE,
    INDEX idx_flight (flight_id),
    INDEX idx_crew (crew_id),
    UNIQUE KEY unique_flight_crew (flight_id, crew_id)
);

-- ============================================
-- TABLE: baggage
-- Baggage tracking
-- ============================================
CREATE TABLE baggage (
    baggage_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    baggage_tag VARCHAR(20) UNIQUE NOT NULL,
    weight_kg DECIMAL(5, 2) NOT NULL,
    baggage_type ENUM('checked', 'carry_on', 'special') NOT NULL,
    status ENUM('checked_in', 'loaded', 'in_transit', 'arrived', 'delayed', 'lost') DEFAULT 'checked_in',
    special_handling VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE CASCADE,
    INDEX idx_booking (booking_id),
    INDEX idx_tag (baggage_tag)
);

-- ============================================
-- TABLE: feedback
-- Customer feedback and ratings
-- ============================================
CREATE TABLE feedback (
    feedback_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    passenger_id INT NOT NULL,
    rating INT CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    category ENUM('service', 'cleanliness', 'punctuality', 'food', 'overall') DEFAULT 'overall',
    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE CASCADE,
    FOREIGN KEY (passenger_id) REFERENCES passengers(passenger_id) ON DELETE CASCADE,
    INDEX idx_rating (rating),
    INDEX idx_submitted (submitted_at)
);

-- ============================================
-- TABLE: contact_messages
-- Customer support messages
-- ============================================
CREATE TABLE contact_messages (
    message_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    subject VARCHAR(200) NOT NULL,
    message TEXT NOT NULL,
    status ENUM('new', 'in_progress', 'resolved', 'closed') DEFAULT 'new',
    priority ENUM('low', 'medium', 'high', 'urgent') DEFAULT 'medium',
    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resolved_at TIMESTAMP NULL,
    INDEX idx_status (status),
    INDEX idx_submitted (submitted_at)
);

-- View: Complete flight information with airport details
CREATE VIEW vw_flight_details AS
SELECT 
    f.flight_id,
    f.flight_number,
    f.departure_time,
    f.arrival_time,
    f.duration_minutes,
    da.airport_code AS departure_code,
    da.city AS departure_city,
    da.country AS departure_country,
    aa.airport_code AS arrival_code,
    aa.city AS arrival_city,
    aa.country AS arrival_country,
    f.economy_price,
    f.business_price,
    f.first_class_price,
    f.available_economy_seats,
    f.available_business_seats,
    f.available_first_class_seats,
    f.status,
    a.aircraft_model,
    a.registration_number
FROM flights f
JOIN airports da ON f.departure_airport_id = da.airport_id
JOIN airports aa ON f.arrival_airport_id = aa.airport_id
JOIN aircraft a ON f.aircraft_id = a.aircraft_id;

-- View: Passenger booking history
CREATE VIEW vw_passenger_bookings AS
SELECT 
    p.passenger_id,
    CONCAT(p.first_name, ' ', p.last_name) AS passenger_name,
    b.booking_reference,
    f.flight_number,
    da.city AS departure_city,
    aa.city AS arrival_city,
    f.departure_time,
    b.class,
    b.seat_number,
    b.total_amount,
    b.payment_status,
    b.booking_status
FROM passengers p
JOIN bookings b ON p.passenger_id = b.passenger_id
JOIN flights f ON b.flight_id = f.flight_id
JOIN airports da ON f.departure_airport_id = da.airport_id
JOIN airports aa ON f.arrival_airport_id = aa.airport_id;

-- View: Revenue summary by flight
CREATE VIEW vw_flight_revenue AS
SELECT 
    f.flight_number,
    COUNT(b.booking_id) AS total_bookings,
    SUM(CASE WHEN b.payment_status = 'completed' THEN b.total_amount ELSE 0 END) AS total_revenue,
    SUM(CASE WHEN b.booking_status = 'confirmed' THEN 1 ELSE 0 END) AS confirmed_bookings,
    da.city AS departure_city,
    aa.city AS arrival_city,
    f.departure_time
FROM flights f
LEFT JOIN bookings b ON f.flight_id = b.flight_id
JOIN airports da ON f.departure_airport_id = da.airport_id
JOIN airports aa ON f.arrival_airport_id = aa.airport_id
GROUP BY f.flight_id;

DELIMITER //

-- Procedure: Book a flight
CREATE PROCEDURE sp_book_flight(
    IN p_passenger_id INT,
    IN p_flight_id INT,
    IN p_class VARCHAR(20),
    IN p_seat_number VARCHAR(5),
    IN p_meal_preference VARCHAR(50)
)
BEGIN
    DECLARE v_price DECIMAL(10, 2);
    DECLARE v_booking_ref VARCHAR(10);
    
    -- Generate booking reference
    SET v_booking_ref = CONCAT('BK', LPAD(FLOOR(RAND() * 999999), 6, '0'));
    
    -- Get price based on class
    SELECT 
        CASE p_class
            WHEN 'economy' THEN economy_price
            WHEN 'business' THEN business_price
            WHEN 'first_class' THEN first_class_price
        END INTO v_price
    FROM flights
    WHERE flight_id = p_flight_id;
    
    -- Insert booking
    INSERT INTO bookings (booking_reference, passenger_id, flight_id, seat_number, class, total_amount, meal_preference)
    VALUES (v_booking_ref, p_passenger_id, p_flight_id, p_seat_number, p_class, v_price, p_meal_preference);
    
    -- Update available seats
    UPDATE flights
    SET available_economy_seats = CASE WHEN p_class = 'economy' THEN available_economy_seats - 1 ELSE available_economy_seats END,
        available_business_seats = CASE WHEN p_class = 'business' THEN available_business_seats - 1 ELSE available_business_seats END,
        available_first_class_seats = CASE WHEN p_class = 'first_class' THEN available_first_class_seats - 1 ELSE available_first_class_seats END
    WHERE flight_id = p_flight_id;
    
    SELECT v_booking_ref AS booking_reference, v_price AS amount;
END //

-- Procedure: Add miles to passenger account
CREATE PROCEDURE sp_add_miles(
    IN p_passenger_id INT,
    IN p_miles INT,
    IN p_booking_id INT,
    IN p_description VARCHAR(255)
)
BEGIN
    -- Update miles balance
    UPDATE frequent_flyer_miles
    SET total_miles = total_miles + p_miles,
        miles_earned = miles_earned + p_miles,
        lifetime_miles = lifetime_miles + p_miles
    WHERE passenger_id = p_passenger_id;
    
    -- Record transaction
    INSERT INTO miles_transactions (passenger_id, booking_id, transaction_type, miles_amount, description)
    VALUES (p_passenger_id, p_booking_id, 'earned', p_miles, p_description);
    
    -- Update tier based on lifetime miles
    UPDATE frequent_flyer_miles
    SET tier = CASE
        WHEN lifetime_miles >= 75000 THEN 'Platinum'
        WHEN lifetime_miles >= 25000 THEN 'Gold'
        ELSE 'Silver'
    END
    WHERE passenger_id = p_passenger_id;
END //

-- Procedure: Cancel booking
CREATE PROCEDURE sp_cancel_booking(
    IN p_booking_id INT
)
BEGIN
    DECLARE v_flight_id INT;
    DECLARE v_class VARCHAR(20);
    
    -- Get booking details
    SELECT flight_id, class INTO v_flight_id, v_class
    FROM bookings
    WHERE booking_id = p_booking_id;
    
    -- Update booking status
    UPDATE bookings
    SET booking_status = 'cancelled', payment_status = 'refunded'
    WHERE booking_id = p_booking_id;
    
    -- Restore available seats
    UPDATE flights
    SET available_economy_seats = CASE WHEN v_class = 'economy' THEN available_economy_seats + 1 ELSE available_economy_seats END,
        available_business_seats = CASE WHEN v_class = 'business' THEN available_business_seats + 1 ELSE available_business_seats END,
        available_first_class_seats = CASE WHEN v_class = 'first_class' THEN available_first_class_seats + 1 ELSE available_first_class_seats END
    WHERE flight_id = v_flight_id;
END //

DELIMITER ;

DELIMITER //

-- Trigger: Automatically add miles when booking is confirmed
CREATE TRIGGER trg_add_miles_on_booking
AFTER UPDATE ON bookings
FOR EACH ROW
BEGIN
    IF NEW.payment_status = 'completed' AND OLD.payment_status != 'completed' THEN
        INSERT INTO miles_transactions (passenger_id, booking_id, transaction_type, miles_amount, description)
        VALUES (NEW.passenger_id, NEW.booking_id, 'earned', FLOOR(NEW.total_amount), 
                CONCAT('Miles earned from booking ', NEW.booking_reference));
        
        UPDATE frequent_flyer_miles
        SET total_miles = total_miles + FLOOR(NEW.total_amount),
            miles_earned = miles_earned + FLOOR(NEW.total_amount),
            lifetime_miles = lifetime_miles + FLOOR(NEW.total_amount)
        WHERE passenger_id = NEW.passenger_id;
    END IF;
END //

-- Trigger: Update offer usage count
CREATE TRIGGER trg_update_offer_usage
AFTER INSERT ON offer_usage
FOR EACH ROW
BEGIN
    UPDATE offers
    SET current_usage = current_usage + 1
    WHERE offer_id = NEW.offer_id;
END //

DELIMITER ;
